# my_robot: a juiz component written in C++ (juiz docs/spec/pkg.md)
#   scripts/dev.sh builds it and puts the plugin and libjuiz_sdk in app/plugins/<target>/
cmake_minimum_required(VERSION 3.16)
project(my_robot VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

# libjuiz_sdk must be the same version as juiz (juiz --version): juiz does not load a plugin built with another version.
# Change the directory with: cmake -D JUIZ_SDK_DIR=<dir> (or the JUIZ_SDK_DIR environment variable of scripts/dev.sh)
set(JUIZ_VERSION "<version>")
if(WIN32)
  file(TO_CMAKE_PATH "$ENV{USERPROFILE}/.juiz/sdk/${JUIZ_VERSION}" JUIZ_SDK_DEFAULT_DIR)
else()
  set(JUIZ_SDK_DEFAULT_DIR "$ENV{HOME}/.juiz/sdk/${JUIZ_VERSION}")
endif()
set(JUIZ_SDK_DEFAULT_DIR "/opt/juiz-sdk")  # given with juiz pkg create --juiz-sdk-dir
set(JUIZ_SDK_DIR "${JUIZ_SDK_DEFAULT_DIR}" CACHE PATH "libjuiz_sdk directory")
find_package(juiz_sdk ${JUIZ_VERSION} EXACT CONFIG REQUIRED PATHS "${JUIZ_SDK_DIR}" NO_DEFAULT_PATH)

# The plugin (libmy_robot.so, libmy_robot.dylib or my_robot.dll) goes to build/plugins.
# $<1:...> keeps Visual Studio from adding the configuration name (build/plugins/Release)
add_library(my_robot SHARED src/my_robot.cpp)
target_link_libraries(my_robot PRIVATE juiz_sdk::juiz_sdk)
set_target_properties(my_robot PROPERTIES
  LIBRARY_OUTPUT_DIRECTORY "$<1:${CMAKE_BINARY_DIR}/plugins>"
  RUNTIME_OUTPUT_DIRECTORY "$<1:${CMAKE_BINARY_DIR}/plugins>")
# Make the plugin look for libjuiz_sdk next to itself ($ORIGIN / @loader_path) and copy libjuiz_sdk
# (juiz_sdk.dll on Windows) to build/plugins, so that the directory works in a .juizapp as is
juiz_sdk_bundle(my_robot)

# The C++ types (record, enum) of the .jidl: juiz idl -g cpp generates juiz_idl.h in build/juiz_idl when the .jidl changes.
# It uses juiz-idl, or `juiz idl` of the juiz in PATH. DEPENDS lists the imported .jidl files
juiz_generate_types(my_robot
  IDL idl/robot.jidl
  DEPENDS idl/common/goal.jidl
)
==== README.md ====
# my_robot

A juiz component written in C++. It was made with `juiz pkg create --idl` of juiz <version>,
from the process / container declarations of `idl/robot.jidl`.


## What is inside

| Type | Kind | Arguments (default) | Returns | What it does |
| --- | --- | --- | --- | --- |
| `turn` | Process | `goal: robot.common.Goal`, `pose: juiz_std.geometry.Pose2D`, `speed: double` (1.0) | `double` | Turns |
| `wheel` | Container | - | - | A wheel |
| `spin` | ContainerProcess | `n: long` (1), `mode: robot.common.Mode` ("FAST") | `long` | - |
| `stop` | ContainerProcess | - | `null` | - |

| Path | What it is |
| --- | --- |
| `idl/robot.jidl` | The types and the declarations (JIDL) |
| `src/my_robot.cpp` | The Component. Each `COMPONENT_*_FACTORY` is a type, listed in `my_robot_component_manifest()`. Write the bodies (`throw`) |
| `CMakeLists.txt` | Builds the plugin with libjuiz_sdk (`find_package(juiz_sdk <version> EXACT)`), and generates the C++ types of the `.jidl` (`juiz_generate_types`, `juiz_idl.h`) |
| `app/app.conf` | The configuration to start juiz with. It is also the entry of the `.juizapp` package |
| `app/juizapp.yaml` | The `.juizapp` manifest (name, version, required juiz version) |
| `scripts/dev.sh`, `scripts/dev.ps1` | Build the plugin and put it with libjuiz_sdk in `app/plugins/<target>/` |
| `scripts/pack.sh`, `scripts/pack.ps1` | Build in release mode and make `dist/my_robot-<version>.juizapp` |

## Requirements

- juiz <version> (`juiz --version`). Install it with `cargo install juiz_app --version <version>` or `pip install juiz-cli==<version>`
- juiz in PATH when you build (CMake runs `juiz idl` to generate the C++ types of the `.jidl`)
- CMake 3.16 or later and a C++17 compiler (on Windows, Visual Studio 2022 with "Desktop development with C++")
- libjuiz_sdk <version> in `~/.juiz/sdk/<version>` (`%USERPROFILE%\.juiz\sdk\<version>` on Windows).
  Install it with `juiz sdk install` (it downloads libjuiz_sdk of the same version as juiz;
  `juiz sdk path` prints where it is).

libjuiz_sdk must be the same version as juiz: juiz does not load a plugin that carries another version of libjuiz_sdk.
`CMakeLists.txt` asks for exactly `<version>`, and `app/juizapp.yaml` requires `juiz: "=<version>"` for the same reason.
When you update juiz, install libjuiz_sdk of the new version and change both.
To use libjuiz_sdk in another directory, run `JUIZ_SDK_DIR=<dir> scripts/dev.sh` (or `cmake -D JUIZ_SDK_DIR=<dir>`).

## Write, build, run and call

The functions in `src/my_robot.cpp` have the arguments and the return values of the `.jidl`, and their bodies throw
`std::runtime_error`: calling one before you write it fails. The example calls below use the default values, or a value of the type.

```sh
# once: put libjuiz_sdk <version> in /opt/juiz-sdk
juiz sdk install --dir /opt/juiz-sdk
# write the bodies of the functions (throw std::runtime_error) in src/my_robot.cpp, then
scripts/dev.sh                  # build and put the plugin and libjuiz_sdk in app/plugins/<target>/
juiz -f app/app.conf -d         # start (keep it running)

# in another terminal
juiz process call "http://127.0.0.1:8000/process/turn0::turn" '{"goal":{"mode":"SLOW","pose":{"position":{"x":0.0,"y":0.0},"theta":0.0},"tolerance":0.1},"pose":{"position":{"x":0.0,"y":0.0},"theta":0.0},"speed":1.0}' -p
juiz container-process call "http://127.0.0.1:8000/container_process/spin0::spin:wheel0" '{"mode":"FAST","n":1}' -p
juiz container-process call "http://127.0.0.1:8000/container_process/stop0::stop:wheel0" '{}' -p

scripts/pack.sh                 # make dist/my_robot-0.1.0.juizapp
juiz run dist/my_robot-0.1.0.juizapp
```

On Windows, use `scripts\dev.ps1` and `scripts\pack.ps1` instead of the `.sh` scripts.

## Changing the .jidl

The build generates the C++ types of `idl/robot.jidl` again when it changes (with `juiz idl`, so juiz must be in PATH when you build).
When you change the declarations, change the functions and the manifests in `src/my_robot.cpp` and `app/app.conf` to match
(`juiz idl -g cpp-skeleton -o <dir> idl/robot.jidl` writes the functions of all the declarations again, to copy from).
When the `.jidl` imports another file, add it to `DEPENDS` of `juiz_generate_types` in `CMakeLists.txt`.

## Versions

The package version is `0.1.0` in both `CMakeLists.txt` and `app/juizapp.yaml`. When you release a new version, change both.

## Plugins for several OSes in one package

Run `scripts/dev.sh --release` on each OS, collect the `app/plugins/<target>/` directories in one `app/plugins/`, and run `scripts/pack.sh --no-build`.
Each `app/plugins/<target>/` holds the plugin and libjuiz_sdk of that OS.

==== scripts/dev.ps1 ====
# Build the plugin and put it with juiz_sdk.dll in app\plugins\<target>\ (to run app\app.conf during development)
# usage: scripts\dev.ps1 [-Release]
#   $env:JUIZ_SDK_DIR = "<dir>" uses libjuiz_sdk in <dir> (default: %USERPROFILE%\.juiz\sdk\<version>)
param([switch]$Release)
$ErrorActionPreference = "Stop"
Set-Location (Join-Path $PSScriptRoot "..")
if (-not (Get-Command juiz -ErrorAction SilentlyContinue)) {
    Write-Host "error: juiz is not in PATH. Install juiz <version>:"
    Write-Host "  cargo install juiz_app --version <version>   (or: pip install juiz-cli==<version>)"
    exit 1
}
if (-not (Get-Command cmake -ErrorAction SilentlyContinue)) {
    Write-Host "error: cmake is not in PATH. Install CMake 3.16 or later and Visual Studio 2022 (Desktop development with C++)."
    exit 1
}
$target = (juiz pkg target | Out-String).Trim()
$config = if ($Release) { "Release" } else { "Debug" }
$cmakeArgs = @("-S", ".", "-B", "build", "-D", "CMAKE_BUILD_TYPE=$config")
if ($env:JUIZ_SDK_DIR) { $cmakeArgs += @("-D", "JUIZ_SDK_DIR=$env:JUIZ_SDK_DIR") }
cmake @cmakeArgs
if ($LASTEXITCODE -ne 0) { exit $LASTEXITCODE }
cmake --build build --config $config
if ($LASTEXITCODE -ne 0) { exit $LASTEXITCODE }
$dest = Join-Path "app\plugins" $target
New-Item -ItemType Directory -Force -Path $dest | Out-Null
foreach ($lib in @("my_robot.dll", "juiz_sdk.dll")) {
    Copy-Item (Join-Path "build\plugins" $lib) $dest
    Write-Output (Join-Path $dest $lib)
}
==== scripts/dev.sh ====
#!/bin/sh
# Build the plugin and put it with libjuiz_sdk in app/plugins/<target>/ (to run app/app.conf during development)
# usage: scripts/dev.sh [--release]
#   JUIZ_SDK_DIR=<dir> scripts/dev.sh   uses libjuiz_sdk in <dir> (default: ~/.juiz/sdk/<version>)
set -eu
cd "$(dirname "$0")/.."
if ! command -v juiz >/dev/null 2>&1; then
  echo "error: juiz is not in PATH. Install juiz <version>:" >&2
  echo "  cargo install juiz_app --version <version>   (or: pip install juiz-cli==<version>)" >&2
  exit 1
fi
if ! command -v cmake >/dev/null 2>&1; then
  echo "error: cmake is not in PATH. Install CMake 3.16 or later and a C++17 compiler." >&2
  exit 1
fi
target=$(juiz pkg target)
if [ "${1:-}" = "--release" ]; then
  config=Release
else
  config=Debug
fi
if [ -n "${JUIZ_SDK_DIR:-}" ]; then
  cmake -S . -B build -D CMAKE_BUILD_TYPE=$config -D "JUIZ_SDK_DIR=$JUIZ_SDK_DIR"
else
  cmake -S . -B build -D CMAKE_BUILD_TYPE=$config
fi
cmake --build build --config $config
case "$target" in
  *apple-darwin) libs="libmy_robot.dylib libjuiz_sdk.dylib" ;;
  *windows*) libs="my_robot.dll juiz_sdk.dll" ;;
  *) libs="libmy_robot.so libjuiz_sdk.so" ;;
esac
mkdir -p "app/plugins/$target"
for lib in $libs; do
  cp "build/plugins/$lib" "app/plugins/$target/"
  echo "app/plugins/$target/$lib"
done
==== scripts/pack.ps1 ====
# Make dist\my_robot-<version>.juizapp with juiz pack (juiz docs/spec/app-package.md)
# usage: scripts\pack.ps1 [-NoBuild]
#   -NoBuild: do not build; pack the plugins already in app\plugins\<target>\
param([switch]$NoBuild)
$ErrorActionPreference = "Stop"
Set-Location (Join-Path $PSScriptRoot "..")
if (-not $NoBuild) {
    & (Join-Path $PSScriptRoot "dev.ps1") -Release
    if ($LASTEXITCODE -ne 0) { exit $LASTEXITCODE }
}
$stage = Join-Path ([System.IO.Path]::GetTempPath()) ("juiz-pack-" + [System.Guid]::NewGuid())
New-Item -ItemType Directory -Path $stage | Out-Null
try {
    Copy-Item -Recurse -Path "app\*" -Destination $stage
    Get-ChildItem -Path $stage -Recurse -Directory -Filter __pycache__ | Remove-Item -Recurse -Force
    foreach ($f in @("LICENSE-MIT", "LICENSE-APACHE")) {
        if (Test-Path $f) { Copy-Item $f $stage }
    }
    New-Item -ItemType Directory -Force -Path dist | Out-Null
    juiz pack $stage -o dist
    if ($LASTEXITCODE -ne 0) { exit $LASTEXITCODE }
} finally {
    Remove-Item -Recurse -Force $stage
}
==== scripts/pack.sh ====
#!/bin/sh
# Make dist/my_robot-<version>.juizapp with juiz pack (juiz docs/spec/app-package.md)
# usage: scripts/pack.sh [--no-build]
#   --no-build: do not build; pack the plugins already in app/plugins/<target>/
#               (to put plugins built on several OSes into one .juizapp, collect them there first)
set -eu
cd "$(dirname "$0")/.."
if [ "${1:-}" != "--no-build" ]; then
  scripts/dev.sh --release
fi
stage=$(mktemp -d)
trap 'rm -rf "$stage"' EXIT
(cd app && tar cf - --exclude=__pycache__ .) | (cd "$stage" && tar xf -)
for f in LICENSE-MIT LICENSE-APACHE; do
  if [ -f "$f" ]; then cp "$f" "$stage/"; fi
done
mkdir -p dist
juiz pack "$stage" -o dist
==== src/my_robot.cpp ====
// my_robot: a juiz component made from idl/robot.jidl (juiz pkg create --idl).
//
// Write the body of each function (replace the `(void)…;` and `throw` lines).
// The arguments and the return values are typed by the .jidl: the record and enum types are
// in juiz_idl.h, which CMake generates from idl/robot.jidl (juiz_generate_types in CMakeLists.txt).
// When you change idl/robot.jidl, change the functions and the manifests here to match.

#include <cstdint>
#include <optional>
#include <stdexcept>
#include <string>
#include <vector>

#include "juiz/juiz.h"
#include "juiz_idl.h"

// Turns
static auto turn_manif = ProcessManifest("turn")
    .description("Turns")
    .add_typed_arg<::robot::common::Goal>("goal", "")
    .add_typed_arg<::juiz_std::geometry::Pose2D>("pose", "")
    .add_float_arg("speed", "", 1.0);
std::optional<double> turn(const ::robot::common::Goal& goal, const ::juiz_std::geometry::Pose2D& pose, double speed) {
    (void)goal;
    (void)pose;
    (void)speed;
    throw std::runtime_error("write the body of turn");
}
COMPONENT_PROCESS_FACTORY(turn_manif, turn);

// State of the container wheel (the JIDL does not type it: add the members you need)
struct WheelState {
};

// A wheel
static auto wheel_manif = ContainerManifest("wheel")
    .description("A wheel");
WheelState* wheel(juiz::CapsuleMap args) {
    (void)args;
    return new WheelState();
}
COMPONENT_CONTAINER_FACTORY(wheel_manif, wheel);

static auto spin_manif = ProcessManifest("spin")
    .container_type("wheel")
    .add_int_arg("n", "", INT64_C(1))
    .add_typed_arg_value<::robot::common::Mode>("mode", "", juiz::Value(std::string("FAST")));
std::optional<int64_t> spin(WheelState* container, int64_t n, ::robot::common::Mode mode) {
    (void)container;
    (void)n;
    (void)mode;
    throw std::runtime_error("write the body of spin");
}
COMPONENT_CONTAINER_PROCESS_FACTORY(WheelState, spin_manif, spin);

static auto stop_manif = ProcessManifest("stop")
    .container_type("wheel");
std::optional<juiz::Value> stop(WheelState* container) {
    (void)container;
    throw std::runtime_error("write the body of stop");
}
COMPONENT_CONTAINER_PROCESS_FACTORY(WheelState, stop_manif, stop);

// The manifest of the whole Component (the types in this plugin)
auto my_robot_component_manifest() {
    return ComponentManifest("my_robot")
        .add_process(turn_manifest())
        .add_container(wheel_manifest()
            .add_process(spin_manifest())
            .add_process(stop_manifest()));
}
DEFINE_COMPONENT_MANIFEST_ENTRY_POINT(my_robot_component_manifest)
